\begin{frame}{Task: normalize a full camera video into screen coordinates}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.61\textwidth}

      \vspace{0.1cm}

        \smalltag{rectified screen-coordinate output}
      \end{center}
    \end{column}
    \begin{column}{0.36\textwidth}
      \begin{block}{Input}
        Hand-held video of a computer screen in a full scene.
      \end{block}
      \begin{block}{Output}
        Front-facing screen-content video with aspect ratio preserved.
      \end{block}
      \begin{itemize}\small
        \item Suppress background.
        \item Correct perspective.
        \item Reduce jitter.
        \item Preserve real content motion.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Current work: preliminary stability check}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.64\textwidth}
      \imageorbox{assets/method_ablation_summary_table.pdf}{\linewidth}{4.55cm}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{Implemented so far}
        Detection, border-guided tracking, homography warping, and residual-motion analysis.
      \end{block}
      \begin{itemize}\small
        \item Frame-wise detection is sensitive to corner jitter.
        \item Content optical flow can follow scrolling or in-screen video.
        \item Border-guided tracking is currently most stable on the pilot clip.
      \end{itemize}
      \talknote{Pilot evidence only; full self-collected dataset evaluation is still planned.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Experiments: self-collected scenarios and baselines}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.60\textwidth}
      \imageorbox{assets/self_collected_dataset_plan.pdf}{\linewidth}{4.55cm}
    \end{column}
    \begin{column}{0.37\textwidth}
      \begin{block}{Dataset plan}
        5 scenario classes, 10 clips per class, about 5 seconds per clip.
      \end{block}
      \begin{itemize}\small
        \item Static pages.
        \item Scrolling pages.
        \item In-screen video playback.
        \item PPT or weak-border pages.
        \item Glare, moire, or partial screen loss.
      \end{itemize}
      \begin{block}{Baselines}
        Frame-wise detection, content optical flow, and proposed border-guided tracking.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
